# CEMS 工业数据采集链路 - Python 依赖
# 安装：pip install -r requirements.txt

# 说明：这里锁了主版本上限，保证"本机运行"和"容器运行"用的是同一套 API。
# 例：pymodbus 3.9 起把 ModbusSlaveContext 改名、slave= 参数改成 device_id=，
#     设备层会直接 ImportError，所以限定在 3.8 及以下。

# 设备层：Modbus TCP
# 下限写 3.6.1 而不是 3.6.0：PyPI 上根本没有 3.6.0（3.5.4 之后直接跳到 3.6.1），
# 写了也会被 pip 解析到 3.6.1+，属于"看着有下限、实际不存在"。
# 实测：3.6.1 下设备层启动 + 网关读 8 个寄存器全部正常。
pymodbus>=3.6.1,<3.9.0

# 消息层：MQTT
paho-mqtt>=2.1.0,<3.0.0

# 存储层：TDengine
taospy>=2.8.0,<3.0.0

# Web层：Flask
flask>=3.0.0,<4.0.0

# Web 层：生产级 WSGI 服务器（Flask 自带的开发服务器不适合对外提供服务）
waitress>=3.0.0,<4.0.0

# Web 层：报表导出 Excel（只写不读，用 write_only 模式流式写）
openpyxl>=3.1,<4.0

# Web 层：查询结果缓存客户端（只缓存"已闭合时间窗"的聚合结果，见 src/web/cache.py）
# 上限锁 5.x：本项目只用 get/set/del/incr/scan，锁上限是为了让"本机运行"与"容器运行"
# 落在同一套 API 上；Redis 服务端不可用时它会整条降级为"直连查库"，不影响功能。
redis>=5.0,<6.0

# 北向协议层：HJ 212 SM4 数据段加解密（仅 src/protocol/hj212/crypto.py 使用）
# 说明：只用它的 SM4 单分组原语，**不用** CryptSM4.crypt_ecb ——
#      该方法加密方向会强制补齐到下一个分组（16 字节明文 → 32 字节密文），
#      与 HJ 212—2025 附录 A.2"不足 16 个字符的部分使用明文"冲突。
#      实测与处理方式见 docs/reference/hj212-协议实现说明.md §5.2。
# 注意：gmssl 自身依赖 pycryptodomex（pip 自动安装），属传递依赖，
#      不是本项目主动引入的第二个库。
gmssl>=3.2.2,<4.0
